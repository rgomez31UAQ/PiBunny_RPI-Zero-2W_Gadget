# SPI for the Waveshare 1.3inch LCD HAT
dtparam=spi=on
# HAT keys/joystick pull-ups come from libgpiod BIAS_PULL_UP in hardware.py

# Keep the LCD backlight OFF while the OS boots (the uninitialized panel
# shows white with the HAT backlight on = the "long white" boot).  bunny-ui
# turns it on right after initializing the display.
